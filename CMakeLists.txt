cmake_minimum_required(VERSION 3.21)
project(nova-desktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Test)
qt_add_executable(nova-desktop src/main.cpp src/systeminfo.cpp src/systeminfo.h)
qt_add_resources(nova-desktop "qml" PREFIX "/" FILES qml/Main.qml)
target_link_libraries(nova-desktop PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
install(TARGETS nova-desktop RUNTIME DESTINATION bin)
enable_testing()
add_executable(nova-systeminfo-test tests/systeminfo_test.cpp src/systeminfo.cpp src/systeminfo.h)
target_link_libraries(nova-systeminfo-test PRIVATE Qt6::Core Qt6::Test)
add_test(NAME nova-systeminfo-test COMMAND nova-systeminfo-test)
